# v0.4 Security Assessment Engine Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the v0.3 discovery prototype into a modular, evidence-driven, unauthenticated security assessment engine with a stable scan-session API, structured findings, stronger authentication/security-control observations, and a new `scan <url>` command.

**Architecture:** Keep the existing v0.3 network, HTML, local, auth-map, and shell modules reusable. Add focused `core` components for target validation, evidence/finding/session models, module registration, and orchestration; expose those components through an engine-backed `scan` command and JSON reporting without collecting credentials or submitting authentication requests.

**Tech Stack:** Python 3.11+, standard library (`dataclasses`, `datetime`, `urllib`, `http.cookies`, `re`, `uuid`, `typing`), pytest 8+, existing `setuptools` package layout, GitHub Actions for reproducible CI.

**Spec:** `docs/superpowers/specs/2026-09-17-v0.4-security-assessment-engine-design.md`

## Global Constraints

- HTTP and HTTPS only.
- GET-based public inspection only.
- No credential-bearing requests.
- No custom cookies supplied by the user.
- No request body submission.
- Bounded response size.
- Bounded redirect count.
- Clear timeout.
- No password testing, cracking, credential stuffing, login automation, token harvesting, exploitation, persistence, destructive actions, or address-range scanning.
- Cookie values must never be stored in scan-session models or reports.
- Existing v0.3 tests must remain passing.
- New modules must be registerable without modifying the core execution loop.

```python
from __future__ import annotations

from dataclasses import dataclass
from urllib.parse import urlparse


@dataclass(frozen=True, slots=True)
class TargetPolicy:
    timeout_seconds: float = 10.0
    max_response_bytes: int = 512 * 1024
    max_redirects: int = 10
    allow_http: bool = True
    allow_https: bool = True


@dataclass(frozen=True, slots=True)
class Target:
    original_input: str
    url: str
    scheme: str
    hostname: str
    port: int | None
    path: str


def normalize_target(value: str) -> Target:
    original = value.strip()
    if not original:
        raise ValueError("Target URL is empty")
    parsed = urlparse(original if "://" in original else f"https://{original}")
    if not parsed.hostname:
        raise ValueError("Target URL must contain a hostname")
    scheme = parsed.scheme.lower()
    path = parsed.path or "/"
    url = parsed._replace(scheme=scheme, path=path).geturl()
    return Target(original, url, scheme, parsed.hostname, parsed.port, path)


def validate_target(target: Target, policy: TargetPolicy | None = None) -> None:
    policy = policy or TargetPolicy()
    if target.scheme not in {"http", "https"}:
        raise ValueError(f"Unsupported URL scheme: {target.scheme}")
    if target.scheme == "http" and not policy.allow_http:
        raise ValueError("HTTP targets are disabled by policy")
    if target.scheme == "https" and not policy.allow_https:
        raise ValueError("HTTPS targets are disabled by policy")
    if not target.hostname:
        raise ValueError("Target hostname is required")
    if policy.timeout_seconds <= 0:
        raise ValueError("Timeout must be positive")
    if policy.max_response_bytes <= 0:
        raise ValueError("Maximum response size must be positive")
    if policy.max_redirects < 0:
        raise ValueError("Maximum redirects cannot be negative")
```

```python
from __future__ import annotations

from dataclasses import asdict, dataclass, field
from datetime import datetime
from typing import Any


def _safe(value: Any) -> Any:
    if isinstance(value, datetime):
        return value.isoformat()
    if isinstance(value, list):
        return [_safe(item) for item in value]
    if isinstance(value, dict):
        return {str(key): _safe(item) for key, item in value.items()}
    if hasattr(value, "to_dict"):
        return value.to_dict()
    return value


@dataclass(slots=True)
class Evidence:
    id: str
    source_module: str
    evidence_type: str
    value: str
    target_url: str | None = None
    metadata: dict[str, Any] = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return _safe(asdict(self))


@dataclass(slots=True)
class Finding:
    id: str
    title: str
    severity: str
    category: str
    summary: str
    evidence_refs: list[str] = field(default_factory=list)
    recommendation: str = ""
    confidence: str = "medium"

    def to_dict(self) -> dict[str, Any]:
        return _safe(asdict(self))


@dataclass(slots=True)
class ModuleResult:
    module: str
    success: bool
    evidence: list[Evidence] = field(default_factory=list)
    observations: dict[str, Any] = field(default_factory=dict)
    findings: list[Finding] = field(default_factory=list)
    error: str | None = None

    def to_dict(self) -> dict[str, Any]:
        return _safe(asdict(self))


@dataclass(slots=True)
class ScanSession:
    session_id: str
    agent_version: str
    started_at: datetime
    target: dict[str, Any]
    policy: dict[str, Any]
    modules: list[ModuleResult] = field(default_factory=list)
    evidence: list[Evidence] = field(default_factory=list)
    findings: list[Finding] = field(default_factory=list)
    errors: list[dict[str, str]] = field(default_factory=list)
    warnings: list[str] = field(default_factory=list)
    ended_at: datetime | None = None

    def to_dict(self) -> dict[str, Any]:
        return _safe(asdict(self))
```

```python
from __future__ import annotations

from typing import Any, Protocol

from .models import ModuleResult
from .target import Target


class DiscoveryModule(Protocol):
    name: str
    def run(self, target: Target, context: dict[str, Any]) -> ModuleResult: ...


class ModuleRegistry:
    def __init__(self) -> None:
        self._modules: dict[str, DiscoveryModule] = {}

    def register(self, module: DiscoveryModule) -> None:
        if not module.name:
            raise ValueError("Module name is required")
        if module.name in self._modules:
            raise ValueError(f"Module already registered: {module.name}")
        self._modules[module.name] = module

    def modules(self) -> tuple[DiscoveryModule, ...]:
        return tuple(self._modules[name] for name in sorted(self._modules))
```

```python
from __future__ import annotations

from http.cookies import SimpleCookie
from typing import Any

from masterkey_agent.core.models import Evidence, ModuleResult
from masterkey_agent.models import NetworkObservation


SAFE_COOKIE_ATTRIBUTES = {"secure", "httponly", "samesite", "path", "domain", "max-age", "expires"}


def parse_set_cookie_attributes(header_value: str) -> dict[str, str | bool]:
    cookie = SimpleCookie()
    cookie.load(header_value)
    parsed: dict[str, str | bool] = {}
    for morsel in cookie.values():
        for key in SAFE_COOKIE_ATTRIBUTES:
            value = morsel[key]
            if key in {"secure", "httponly"}:
                if value != "":
                    parsed[key] = True
            elif value:
                parsed[key] = value
    return parsed


class SecurityControlsModule:
    name = "security_controls"

    def run(self, target, context: dict[str, Any]) -> ModuleResult:
        observation: NetworkObservation | None = context.get("network")
        if observation is None:
            return ModuleResult(self.name, False, error="network observation missing")
        evidence: list[Evidence] = []
        headers = {key.lower(): value for key, value in observation.headers.items()}
        mappings = {
            "strict-transport-security": ("header.hsts", "HSTS header present"),
            "content-security-policy": ("header.csp", "CSP header present"),
            "x-content-type-options": ("header.x_content_type_options", "X-Content-Type-Options header present"),
            "referrer-policy": ("header.referrer_policy", "Referrer-Policy header present"),
            "permissions-policy": ("header.permissions_policy", "Permissions-Policy header present"),
            "x-frame-options": ("header.x_frame_options", "X-Frame-Options header present"),
        }
        for header, (kind, description) in mappings.items():
            if header in headers:
                evidence.append(Evidence(f"{kind}-{len(evidence)+1}", self.name, kind, description, target.url))
        for raw_cookie in observation.headers.get("set-cookie", "").split("\n"):
            attrs = parse_set_cookie_attributes(raw_cookie)
            for attribute in ("secure", "httponly", "samesite"):
                if attribute in attrs:
                    evidence.append(
                        Evidence(
                            f"cookie-{attribute}-{len(evidence)+1}",
                            self.name,
                            f"cookie.{attribute}_attribute",
                            "present" if attrs[attribute] is True else str(attrs[attribute]),
                            target.url,
                        )
                    )
        return ModuleResult(self.name, True, evidence=evidence, observations={"controls_observed": len(evidence)})


def observe_security_controls(observation: NetworkObservation, target) -> ModuleResult:
    return SecurityControlsModule().run(target, {"network": observation})
```

Note: `network.py` must normalize multiple `Set-Cookie` response headers into a safe, enumerable representation without persisting raw cookie values. If the HTTP client exposes only a combined header, preserve line-separated cookie headers in a single sanitized field and redact values before returning it.

```python
from __future__ import annotations

from typing import Any
from urllib.parse import urlparse

from masterkey_agent.core.models import Evidence, ModuleResult
from masterkey_agent.core.target import Target
from masterkey_agent.models import NetworkObservation


def _has_marker(value: str, markers: tuple[str, ...]) -> bool:
    lower = value.lower()
    return any(marker in lower for marker in markers)


def analyze_auth_surface(observation: NetworkObservation, target: Target) -> ModuleResult:
    evidence: list[Evidence] = []
    redirects = " ".join(observation.redirects)
    if _has_marker(redirects, ("/oauth", "/authorize", "openid", "oidc")):
        evidence.append(Evidence("auth-oauth-1", "auth_surface", "auth.oauth_redirect_marker", "OAuth/OIDC-like redirect marker observed", target.url))
    if _has_marker(redirects, ("saml", "samlrequest", "samlresponse")):
        evidence.append(Evidence("auth-saml-1", "auth_surface", "auth.saml_redirect_marker", "SAML-like redirect marker observed", target.url))
    challenge = observation.headers.get("www-authenticate", "")
    if challenge:
        evidence.append(Evidence("auth-www-1", "auth_surface", "auth.www_authenticate", "WWW-Authenticate challenge observed", target.url, {"schemes": challenge.split(",")[0]}))
    html = observation.public_html
    if html:
        for index, form in enumerate(html.forms, start=1):
            has_password = any(field.type == "password" or field.autocomplete in {"current-password", "new-password"} for field in form.fields)
            has_identity = any(
                (field.name or "").lower() in {"email", "username", "user", "identifier", "login"}
                or (field.autocomplete or "").lower() == "username"
                for field in form.fields
            )
            if has_password:
                evidence.append(Evidence(f"auth-password-{index}", "auth_surface", "auth.password_field", "Public form exposes password-field metadata", target.url))
            if has_identity:
                evidence.append(Evidence(f"auth-identity-{index}", "auth_surface", "auth.identity_field", "Public form exposes identity-field metadata", target.url))
            if _has_marker(form.action, ("/login", "/signin", "/sign-in", "/auth")):
                evidence.append(Evidence(f"auth-form-{index}", "auth_surface", "auth.login_form", "Public form action appears authentication-oriented", target.url))
        if html.title and _has_marker(html.title, ("sign in", "signin", "log in", "login")):
            evidence.append(Evidence("auth-title-1", "auth_surface", "auth.login_title", "Public page title is authentication-oriented", target.url))
    return ModuleResult("auth_surface", True, evidence=evidence, observations={"evidence_count": len(evidence)})
```

```python
from __future__ import annotations

from datetime import datetime, timezone
from uuid import uuid4

from masterkey_agent import __version__
from masterkey_agent.discovery.network import inspect_url

from .models import ModuleResult, ScanSession
from .registry import ModuleRegistry
from .target import TargetPolicy, normalize_target, validate_target


class ScanEngine:
    def __init__(self, registry: ModuleRegistry, policy: TargetPolicy | None = None, agent_version: str = "0.4.0") -> None:
        self.registry = registry
        self.policy = policy or TargetPolicy()
        self.agent_version = agent_version

    def scan(self, target_input: str) -> ScanSession:
        target = normalize_target(target_input)
        validate_target(target, self.policy)
        started = datetime.now(timezone.utc)
        session = ScanSession(
            session_id=str(uuid4()),
            agent_version=self.agent_version,
            started_at=started,
            target={"original_input": target.original_input, "url": target.url, "scheme": target.scheme, "hostname": target.hostname, "port": target.port, "path": target.path},
            policy={"timeout_seconds": self.policy.timeout_seconds, "max_response_bytes": self.policy.max_response_bytes, "max_redirects": self.policy.max_redirects},
        )
        observation = inspect_url(target.url, timeout=self.policy.timeout_seconds, max_bytes=self.policy.max_response_bytes, max_redirects=self.policy.max_redirects)
        context = {"network": observation}
        session_target = target
        for module in self.registry.modules():
            try:
                result = module.run(session_target, context)
            except Exception as exc:
                result = ModuleResult(module.name, False, error=str(exc))
                session.errors.append({"module": module.name, "stage": "run", "message": str(exc)})
            session.modules.append(result)
            session.evidence.extend(result.evidence)
            session.findings.extend(result.findings)
            if result.error:
                session.errors.append({"module": module.name, "stage": "result", "message": result.error})
        session.ended_at = datetime.now(timezone.utc)
        return session
```

The existing `inspect_url` signature must be extended compatibly so existing callers/tests continue working: new keyword-only bounds should default to the current v0.3 values.

```python
from pathlib import Path

from masterkey_agent.core.models import ScanSession


def write_scan_report(path: str | Path, session: ScanSession) -> None:

        "agent_version": data["agent_version"],
```
